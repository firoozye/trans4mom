% Slide 4: Philosophy of Momentum
% -------------------------------------------------------------
\begin{frame}{The Philosophy of Momentum: Buy High, Sell Higher}
  \begin{block}{Deconstructing the Core Heuristic}
    Standard finance theory posits that asset returns follow a random walk with zero forecastable drift in frictionless markets. Systematic trend following inverts the classical value heuristic: rather than buying low and selling high, trend following executes the rule \emph{buy high and sell higher, or sell short and cover lower}.
  \end{block}
  \vspace{0.1cm}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \textbf{Behavioural Origins}
      \begin{itemize}\scriptsize
        \item \textbf{Disposition effect:} Investors realise gains prematurely while holding losers, causing initial underreaction to fundamental news.
        \item \textbf{Anchoring \& conservatism:} Slow Bayesian updating creates persistent auto-correlated price discovery.
        \item \textbf{Institutional herding:} Benchmark tracking and delegated mandates produce sustained capital flows.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Market Microstructure}
      \begin{itemize}\scriptsize
        \item \textbf{Liquidity provision limits:} Capital constraints among market makers prevent instantaneous price adjustment.
        \item \textbf{Feedback loops:} Systematic portfolio rebalancing and stop-loss execution amplify intermediate directional displacement.
        \item \textbf{Multi-horizon diffusion:} Information diffuses at differing speeds across diverse participant horizons.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% -------------------------------------------------------------
% Slide 5: TSMOM vs. XSMOM
% -------------------------------------------------------------
\begin{frame}{Two Flavours of Momentum: TSMOM vs. XSMOM}
  \begin{block}{Time-Series Momentum (TSMOM / Absolute Momentum)}
    Evaluates an asset strictly against its own historical trajectory:
    \begin{equation*}
      s_{i,t}^{\text{TS}} = \text{sign}\left(r_{i,t}^{(k)}\right), \quad r_{i,t}^{(k)} = \ln\left(P_{i,t}/P_{i,t-k}\right)
    \end{equation*}
    Dominant in managed futures and CTAs across liquid multi-asset derivatives (rates, FX, indices, commodities). Documented by \href{https://doi.org/10.1016/j.jfineco.2011.11.003}{Moskowitz, Ooi, and Pedersen (2012)}.
  \end{block}
  \begin{block}{Cross-Sectional Momentum (XSMOM / Relative Strength)}
    Evaluates an asset relative to the contemporaneous peer return distribution:
    \begin{equation*}
      s_{i,t}^{\text{XS}} = \mathbf{1}_{\{r_{i,t}^{(k)} \in Q_{\text{top}}\}} - \mathbf{1}_{\{r_{i,t}^{(k)} \in Q_{\text{bottom}}\}}
    \end{equation*}
    Dominant in equity factor investing (\href{https://doi.org/10.1111/j.1540-6261.1993.tb04702.x}{Jegadeesh and Titman 1993}, \href{https://doi.org/10.1111/jofi.12021}{Asness, Moskowitz, and Pedersen 2013}).
  \end{block}
\end{frame}

% -------------------------------------------------------------
% Slide 6: Digital Trend Filters
% -------------------------------------------------------------
\begin{frame}{Classical Signal Formulations}
  \begin{block}{Digital Filtering Implementations}
    \begin{itemize}\scriptsize
      \item \textbf{SMA Crossover:} $s_t = \text{sign}\left(\text{SMA}_t(k_{\text{fast}}) - \text{SMA}_t(k_{\text{slow}})\right)$
      \item \textbf{EWMA Crossover:} $s_t = \text{sign}\left(\text{EWMA}_t(\lambda_{\text{fast}}) - \text{EWMA}_t(\lambda_{\text{slow}})\right)$
      \item \textbf{Donchian Channel:} $s_t = +1 \text{ if } P_t = \max_{j} P_{t-j}, \quad -1 \text{ if } P_t = \min_{j} P_{t-j}$
      \item \textbf{MACD Signal:} Normalized difference of exponential moving averages scaled by rolling volatility:
      \begin{equation*}
        \text{MACD}_t(S, L) = \frac{\text{EMA}_t(S) - \text{EMA}_t(L)}{\sigma_t \cdot P_t}
      \end{equation*}
    \end{itemize}
  \end{block}
  \begin{block}{Position Sizing \& Volatility Scaling}
    Traditional trend following sizes contracts inversely proportional to recent rolling volatility:
    \begin{equation*}
      w_{i,t} = \frac{\sigma_{\text{target}}}{\hat{\sigma}_{i,t}} \cdot s_{i,t}
    \end{equation*}
    ensuring risk parity across volatile commodities and low-volatility fixed income.
  \end{block}
\end{frame}

% -------------------------------------------------------------
% Slide 7: Filter Equivalence Theorem
% -------------------------------------------------------------
\begin{frame}{The Filter Equivalence Theorem}
  \begin{block}{Linear Filter Representation (\href{https://ssrn.com/abstract=2465623}{Bruder \& Gaussel 2011}, \href{http://www.thierry-roncalli.com/download/Alternative_Risk_Premia.pdf}{Hamdan et al. 2012})}
    Any linear digital trend filter can be expressed as a convolution kernel of historical returns:
    \begin{equation*}
      s_{t-1} = \sum_{k=1}^\infty \phi_k r_{t-k}
    \end{equation*}
    where $\phi_k$ are the implicit lag weights determined by the filter transfer function.
  \end{block}
  \vspace{0.1cm}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \textbf{Core Equivalence Findings}
      \begin{itemize}\scriptsize
        \item Different digital filters (SMA, EWMA, MACD) yield strategy positions with mutual correlations $>0.90$.
        \item Apparent diversity across moving average heuristics is mathematical redundancy.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Key Takeaway}
      \begin{itemize}\scriptsize
        \item Classical heuristic tweaking (e.g. testing 50/200 vs 40/180) merely shifts lag weights slightly.
        \item A fundamentally different paradigm is needed to capture dynamic regimes.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% Slide 10: Lookback Straddle Equivalence
% -------------------------------------------------------------
\begin{frame}{Mathematical Convexity: Fung-Hsieh Straddle Equivalence}
  \begin{block}{Fung-Hsieh Lookback Option Equivalence (\href{https://people.duke.edu/~dah7/RFS2001.pdf}{Fung \& Hsieh 2001})}
    A continuous-time trend-following strategy with dynamic stop-losses and position flipping generates an asymmetric payoff mathematically equivalent to a portfolio of lookback straddles:
    \begin{equation*}
      \text{Payoff}_{\text{lookback}} = \max_{0 \le t \le T} P_t - \min_{0 \le t \le T} P_t
    \end{equation*}
  \end{block}
  \vspace{0.1cm}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \textbf{Structural Intuition}
      \begin{itemize}\scriptsize
        \item Delivers positive gamma in large directional moves, regardless of market direction.
        \item Trend followers capture extreme market deviations without paying explicit option premiums.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{The Cost of Convexity}
      \begin{itemize}\scriptsize
        \item Whip-saw losses in sideways, range-bound regimes represent the implicit "theta decay" of trend following.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% -------------------------------------------------------------
% Slide 11: Martin-Zou Skewness Theorem
% -------------------------------------------------------------
\begin{frame}{Skewness \& Asymmetric Tail Risk: Martin-Zou Theorem}
  \begin{block}{The Martin-Zou Positive Skewness Theorem (\href{https://www.risk.net/derivatives/structured-products/2194247/momentum-trading-skews-me}{Martin \& Zou 2012})}
    When filtering i.i.d. Gaussian innovations $X_t \sim \mathcal{N}(0, \sigma^2)$ through an EWMA trend filter $s_t = c \sum_{k=1}^\infty \rho^k X_{t-k}$, the compounded strategy return possesses strictly positive skewness:
    \begin{equation*}
      \mathbb{E}\left[R_{t+T}^3\right] = 6 \sum_{j=1}^T (T - j) a_j \gamma_j^a > 0
    \end{equation*}
    Unlike carry and short-volatility strategies that suffer severe negative skewness (tail crashes), momentum delivers structural right skewness.
  \end{block}
  \vspace{-0.1cm}

\end{frame}

\begin{frame}{Factor Momentum: The Explanatory Driver}
  \begin{block}{The Explanatory Factor Finding (\href{https://www.nber.org/papers/w25551}{Ehsani \& Linnainmaa 2022})}
    Individual stock momentum is largely inherited from momentum in underlying systematic factor returns:
    \begin{equation*}
      R_{i,t} = \sum_{k=1}^K \beta_{ik} F_{k,t} + \epsilon_{i,t}
    \end{equation*}
    Cross-sectional equity momentum collapses once controlling for the persistence of systematic style and industry factors.
  \end{block}
  \vspace{-0.1cm}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}

    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Key Implications}
      \begin{itemize}\scriptsize
        \item Momentum is not an idiosyncratic stock anomaly; it is a macro phenomenon driving factor premia.
        \item Factors (Value, Size, Investment, Profitability) exhibit strong auto-regressive momentum.
        \item Supports modeling macro multi-asset universes directly.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% Slide 15: Momentum Crashes
% -------------------------------------------------------------
\begin{frame}{Momentum Crashes: The Rebound Anomaly}
  \begin{block}{The Rebound Crash Anomaly (\href{https://www.nber.org/papers/w20439}{Daniel \& Moskowitz 2016})}
    Unhedged momentum strategies suffer devastating crashes during sudden market inflection points, particularly V-shaped recoveries following panics (e.g. 1932, 2008--2009 GFC). The short leg behaves like a written call option with negative gamma on distressed recovery assets.
  \end{block}
  \vspace{-0.1cm}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \begin{figure}[ht]
        \centering
        \includegraphics[height=3.1cm,keepaspectratio]{L5/XSMomDDGFC.png}
        \caption{Cross-sectional momentum drawdown during the 2008--2009 GFC rebound (Daniel \& Moskowitz 2016).}
      \end{figure}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Crash Mechanism}
      \begin{itemize}\scriptsize
        \item In a market panic, "losers" become highly levered, distressed assets.
        \item When the market abruptly rebounds, distressed losers surge dramatically.
        \item The momentum short leg experiences severe short-squeeze losses.
        \item \emph{Remedy:} Dynamic volatility scaling and regime-aware positioning.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% -------------------------------------------------------------
% Slide 16: Limitations of Classical Heuristics
% -------------------------------------------------------------
\begin{frame}{Limitations of Fixed Heuristics}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \textbf{1. The Lag Drag Dilemma}
      \begin{itemize}\scriptsize
        \item Long lookback windows (e.g., 200 days) protect against high-frequency noise but bleed capital when regimes pivot.
        \item Short windows (e.g., 10 days) adapt rapidly to pivots but suffer severe transaction cost bleed in choppy markets.
      \end{itemize}
      \vspace{0.1cm}
      \textbf{2. Fixed-Threshold Brittleness}
      \begin{itemize}\scriptsize
        \item Discrete decisions ($s_t \in \{-1, +1\}$) ignore continuous regime uncertainty.
        \item Linear scaling fails to model cross-asset volatility spillovers.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{3. Classical State-Space Extensions}
      \begin{itemize}\scriptsize
        \item \textbf{Kalman filtering:} Estimates latent trend slope dynamically, but assumes Gaussian innovations.
        \item \textbf{BOCPD:} Computes run-length hazard rates, but struggles with multi-asset interactions.
      \end{itemize}
      \vspace{0.1cm}
      \begin{alertblock}{Motivation for Deep Learning}
        Can a neural network learn dynamic position sizing end-to-end directly from multi-asset return history?
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Optimizing for the Right Metric: Why MSE Fails}
  \begin{block}{The Breakdown of Mean Squared Error in Trading}
    Supervised learning architectures trained on point forecasting (minimizing Mean Squared Error $\mathcal{L}_{\text{MSE}} = \frac{1}{N}\sum (\hat{r}_{t+1} - r_{t+1})^2$) exhibit near-zero performance when deployed in live strategy execution.
  \end{block}
  \vspace{0.1cm}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \textbf{Why MSE Disconnects from PnL}
      \begin{itemize}\scriptsize
        \item \textbf{Low SNR:} Returns are dominated by noise. A model minimizing MSE converges to predicting zero return everywhere.
        \item \textbf{Symmetric Penalty:} MSE treats missing a rally identically to holding an inverse position in a crash.
        \item \textbf{Directional Indifference:} Small positive predictions for large returns yield low MSE, but fail to size capital.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{The Financial Objective}
      \begin{itemize}\scriptsize
        \item Quant practitioners trade risk-adjusted distributions, not point forecasts.
        \item The network must solve for the optimal policy mapping history to position: $f_\theta: \mathcal{X}_{1:t} \to z_t$.
        \item Loss functions must reflect risk-adjusted portfolio return utility directly:
        \begin{equation*}
          \max_\theta \quad \text{Sharpe}(R_1, \dots, R_T)
        \end{equation*}
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% Slide 19: Differentiable Sharpe Loss
% -------------------------------------------------------------
\begin{frame}{Formulating Differentiable Sharpe Loss}
  \begin{block}{End-to-End Sharpe Loss}
    Let $z_t \in [-1, 1]$ denote position at time $t$, and $r_{t+1}$ the realized volatility-scaled return. The strategy return net of transaction cost $c$ is:
    \begin{equation*}
      R_{t+1} = z_t r_{t+1} - c \cdot |z_t - z_{t-1}|
    \end{equation*}
    Over a training batch sequence of length $B$, we compute empirical mean and variance:
    \begin{equation*}
      \hat{\mu} = \frac{1}{B} \sum_{t=1}^B R_t, \quad \hat{\sigma}^2 = \frac{1}{B-1} \sum_{t=1}^B (R_t - \hat{\mu})^2 + \epsilon
    \end{equation*}
    The loss minimized during gradient backpropagation is the negative annualized Sharpe:
    \begin{equation*}
      \mathcal{L}_{\text{Sharpe}}(\theta) = -\sqrt{252} \cdot \frac{\hat{\mu}}{\hat{\sigma}}
    \end{equation*}
  \end{block}
\end{frame}

% -------------------------------------------------------------
% Slide 20: Differentiable Sortino Loss
% -------------------------------------------------------------
\begin{frame}{Downside Geometry: Differentiable Sortino Loss}
  \begin{block}{Penalizing Downside Semivariance: Sortino Formulation}
    To preserve structural positive return skewness without penalizing upside windfalls, we replace total standard deviation with downside semi-deviation:
    \begin{equation*}
      \hat{\sigma}_{\text{down}}^2 = \frac{1}{B} \sum_{t=1}^B \min(R_t - \tau, 0)^2 + \epsilon
    \end{equation*}
    where $\tau$ is the minimum acceptable return (MAR, typically 0). The Sortino loss is:
    \begin{equation*}
      \mathcal{L}_{\text{Sortino}}(\theta) = -\sqrt{252} \cdot \frac{\hat{\mu} - \tau}{\hat{\sigma}_{\text{down}}}
    \end{equation*}
  \end{block}
  \begin{itemize}\scriptsize
    \item Standard Sharpe penalizes outsized positive trend returns as "volatility".
    \item Sortino loss guides gradient updates to minimize drawdowns while leaving upside trend runs unrestricted.
  \end{itemize}
\end{frame}

% -------------------------------------------------------------
% Slide 21: Turnover Costs & Subgradient Smoothing
% -------------------------------------------------------------
\begin{frame}{Transaction Costs \& Turnover Frictions}
  \begin{block}{Turnover Friction Penalisation}
    Without transaction cost penalties, neural networks over-trade high-frequency noise, producing strategies with high gross Sharpe but deeply negative net Sharpe after slippage.
  \end{block}
  \vspace{0.1cm}
  \begin{block}{Subgradient Smoothing for Turnover Cost}
    Because $|z_t - z_{t-1}|$ has a non-differentiable cusp at zero, production implementations apply smooth approximations:
    \begin{equation*}
      ||\Delta z_t||_{\epsilon} \approx \sqrt{(z_t - z_{t-1})^2 + \epsilon_{\text{smooth}}}
    \end{equation*}
    enabling stable backward passes through AdamW optimizers.
  \end{block}
  \begin{itemize}\scriptsize
    \item Penalizing turnover forces the network to hold positions across micro-reversals unless the regime shift signal is overwhelming.
  \end{itemize}
\end{frame}

import torch
import torch.nn as nn

class DifferentiableSharpeLoss(nn.Module):
    def __init__(self, trans_cost: float = 0.0001, ann_factor: float = 252.0):
        super().__init__()
        self.trans_cost = trans_cost
        self.ann_factor = torch.sqrt(torch.tensor(ann_factor))

    def forward(self, positions: torch.Tensor, returns: torch.Tensor) -> torch.Tensor:
        active_pos = positions[:, :-1]
        active_rets = returns[:, 1:]
        
        # Smooth transaction cost on position delta
        pos_delta = positions[:, 1:] - positions[:, :-1]
        costs = self.trans_cost * torch.sqrt(pos_delta.pow(2) + 1e-8)
        net_returns = (active_pos * active_rets) - costs
        
        mean_ret = torch.mean(net_returns)
        std_ret = torch.std(net_returns, unbiased=True) + 1e-6
        return - (mean_ret / std_ret) * self.ann_factor.to(returns.device)
\end{lstlisting}
\end{frame}

\begin{frame}{Variable Selection Networks (VSN) \& Static Conditioning}
  \begin{block}{Feature Sparsity \& Noise Suppression}

    \begin{equation*}

    \begin{equation*}
      \tilde{x}_t = \sum_{j=1}^M v_t^{(j)} \cdot \text{GRN}_j\left(x_t^{(j)}\right)

  \end{block}
  \begin{itemize}\scriptsize
    \item In calm bull markets, the VSN allocates weight to slow momentum features (e.g., 252-day return, MACD 32/96).
    \item During market shocks, the VSN shifts focus toward fast return windows and Bayesian changepoint hazard metrics.
  \end{itemize}

\begin{frame}{Structural Inductive Bias: The Residual Prior Anchoring}
  \begin{alertblock}{The Overfitting Hazard of Unconstrained Transformers}
    Unconstrained transformers possess enormous parameter capacity ($O(10^5 \text{ to } 10^7)$ weights). In low signal-to-noise financial environments, unguided transformers readily memorize spurious patterns, yielding catastrophic out-of-sample collapse.
  \end{alertblock}
  \begin{block}{The Residual "Handcuffs" Principle}
    Rather than requiring the network to learn basic directional momentum from scratch, the Momentum Transformer anchors its output to a robust classical baseline prior (e.g. equal-weighted multi-scale MACD):
    \begin{equation*}
      \text{Position}_t = \tanh\left(\text{Prior}_t \cdot (1 + \alpha_t)\right)
    \end{equation*}
    where $\alpha_t \in \mathbb{R}$ is the learned residual modulation output.
  \end{block}
  \begin{itemize}\scriptsize
    \item Zero-weight initialization of the final linear projection layer guarantees that $\alpha_t \to 0$ initially: the model starts execution as a proven classical CTA before fine-tuning regime adjustments.
    \item The network learns \emph{when to scale up, size down, or neutralize positions}, avoiding unanchored direction flipping.
  \end{itemize}
\end{frame}

class FeatureProcessor:
    def compute_volatility(self, df: pd.DataFrame) -> pd.DataFrame:
        """60-day EWMA volatility. Computed ex-ante without look-ahead."""
        df['sigma_d'] = df['returns'].ewm(span=self.vol_span).std()
        return df

    def compute_macd_signals(self, df: pd.DataFrame) -> pd.DataFrame:
        for S, L in self.macd_pairs:
            ema_s = df['close'].ewm(span=S).mean()
            ema_l = df['close'].ewm(span=L).mean()
            macd = (ema_s - ema_l) / (df['sigma_d'] * df['close'] + 1e-9)
            macd_std = macd.rolling(252).std() + 1e-9
            df[f'macd_{S}_{L}'] = macd / macd_std
        return df

    def compute_additional_priors(self, df: pd.DataFrame) -> pd.DataFrame:
        """Baseline Prior = Equal-weighted average of multi-scale MACD."""
        macd_cols = [f'macd_{S}_{L}' for S, L in self.macd_pairs]
        df['prior_mom'] = df[macd_cols].mean(axis=1)
        return df
\end{lstlisting}
  \begin{itemize}\tiny
    \item \textbf{Review Note:} Features must never use full-sample normalisation; EWMA volatility and MACD rolling variances must be strictly causal.
  \end{itemize}

class SharpeLoss(nn.Module):
    def forward(self, positions, returns, spreads=None, vol_scales=None):
        # Active positions at t generate return at t+1
        active_pos = positions[:, :-1, :]
        active_rets = returns[:, 1:, :]
        port_returns = (active_pos * active_rets).sum(dim=-1, keepdim=True)
        
        # Scale positions to physical sizing for turnover cost
        physical_pos = positions * vol_scales if vol_scales is not None else positions
        pos_diff = torch.abs(physical_pos - torch.roll(physical_pos, 1, dims=1))
        trading_costs = (self.trans_cost + spreads / 2.0) * pos_diff[:, 1:, :]
        
        # Net portfolio return & Pooled Sharpe across batch
        net_returns = port_returns - trading_costs
        mean_ret = torch.mean(net_returns)
        std_ret = torch.std(net_returns, unbiased=False) + 1e-6
        return - (mean_ret / std_ret) * self.annualization
\end{lstlisting}
  \begin{itemize}\tiny
    \item \textbf{Review Note:} Pooled batch Sharpe prevents single-asset optimization instability and forces cross-asset risk-budget coordination.
  \end{itemize}
\end{frame}

\begin{frame}{Rigorous Backtesting Protocol: Mitigating Look-Ahead Bias}
  \begin{block}{Three Pillars of Academic Backtest Rigor}
    \begin{enumerate}\scriptsize
      \item \textbf{Point-in-Time Feature Alignment:}
      Signals at bar $t$ must only use information available before close $t$. Return realization occurs at $t+1$:
      \begin{equation*}
        \text{PnL}_{t+1} = z_t \cdot r_{t+1} - c \cdot |z_t - z_{t-1}|
      \end{equation*}
      Any normalisation using full-sample moments ($\mu_{\text{all}}, \sigma_{\text{all}}$) introduces catastrophic look-ahead bias.
      \item \textbf{Expanding-Window Walk-Forward Retraining:}
      Rather than a static single train/test split, retrain model parameters periodically (e.g., annually):
      \begin{itemize}\tiny
        \item Train on $[1990, 2015] \to$ Test $2016$
        \item Train on $[1990, 2016] \to$ Test $2017$, etc.
      \end{itemize}
      \item \textbf{Multi-Seed Ensembling:}
      Single transformer runs exhibit substantial random seed variance. Wood et al. average the outputs of a 5-seed ensemble to smooth out idiosyncratic execution timing and boost realized Sharpe ratio by $0.4$--$0.8$.
    \end{enumerate}
  \end{block}
\end{frame}
